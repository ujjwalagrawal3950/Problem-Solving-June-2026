1class Solution {
2    int gcd(int a , int b){
3        return b == 0 ? a : gcd(b, a%b);
4    }
5public:
6    long long maxPairStrength(vector<int>& nums) {
7        int n  = nums.size();
8        long long ans = INT_MIN;
9        for(int i = 0; i<n; i++){
10            for(int j = i+1; j < n; j++){
11                long long product = nums[i];
12                product *= (nums[j]);
13                int x  = gcd(nums[i], nums[j]);
14                ans =    max(ans, (long long)(product/pow(x,2)));
15            }
16        }
17        return ans;
18    }
19};
