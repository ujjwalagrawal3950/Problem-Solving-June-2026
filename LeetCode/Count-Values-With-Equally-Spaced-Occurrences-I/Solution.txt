1class Solution {
2public:
3    int countSpecialIntegers(vector<int>& nums) {
4        int n = nums.size();
5        int count = 0;
6        unordered_map<int, vector<int>>mp;
7        for(int i = 0; i<n; i++){
8            mp[nums[i]].push_back(i);
9        }
10        for(auto it : mp){
11            if(it.second.size() == 3){
12                auto x = it.second;
13                if(x[1] - x[0] == x[2] - x[1]){
14                    count++;
15                }
16            }
17        }
18        return count;
19    }
20};
