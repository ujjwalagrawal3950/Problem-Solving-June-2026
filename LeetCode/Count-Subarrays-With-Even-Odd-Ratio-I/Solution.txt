1class Solution {
2public:
3    int countRatioSubarrays(vector<int>& nums, int a, int b) {

5        int count = 0;
6        for(int i  = 0; i<n; i++){
7            int even = 0 , odd = 0;
8            for(int j = i; j<n; j++){
9                 if(nums[j] % 2 != 0){odd++;}

13        }
14        return count;
15    }
